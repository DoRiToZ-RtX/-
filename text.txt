    <script>
        // State variables
        let currentInput = '0';
        let previousInput = '';
        let operation = null;
        let isPunkMode = false;
        let audioInitialized = false;

        // Display Elements
        const mainDisplay = document.getElementById('mainDisplay');
        const historyDisplay = document.getElementById('historyDisplay');
        const modeStatus = document.getElementById('modeStatus');
        const punkToggle = document.getElementById('punkToggle');

        // Quotes from Boss the 16cm red-eared slider turtle
        const bossQuotes = [
            "Людовик XVI брал микрозаймы, а я просто грею панцирь под лампой 💡",
            "Геометрия пройдет, а 16-сантиметровый пентхаус останется навсегда! 🐢",
            "Сектор Газа в колонках — единственный нормальный фоновый шум в Оренбурге 🎸",
            "Калькулятор в светлых тонах? Ты с ума сошел?! Только темный темы! 🕶️",
            "Зачем деменция, если есть 10 билетов по геометрии? 🧠📐",
            "Куплинов и Витек Play уже всё вычистили, сиди на чиле и пей каркаде ☕"
        ];

        // Sound Effects setup using Tone.js
        let synth, dist;
        function initAudio() {
            if (!audioInitialized) {
                dist = new Tone.Distortion(0.8).toDestination();
                synth = new Tone.PolySynth(Tone.Synth, {
                    oscillator: { type: "sawtooth" },
                    envelope: { attack: 0.01, decay: 0.1, sustain: 0.2, release: 0.2 }
                }).connect(dist);
                audioInitialized = true;
            }
        }

        function playClickSound(freq = "C4") {
            if (isPunkMode) {
                try {
                    initAudio();
                    Tone.start();
                    synth.triggerAttackRelease(freq, "8n");
                } catch(e) {}
            }
        }

        function updateDisplay() {
            mainDisplay.innerText = currentInput;
            if (operation && previousInput !== '') {
                historyDisplay.innerText = `${previousInput} ${getOpSymbol(operation)}`;
            } else {
                historyDisplay.innerText = '';
            }
        }

        function getOpSymbol(op) {
            switch(op) {
                case '+': return '+';
                case '-': return '−';
                case '*': return '×';
                case '/': return '÷';
                default: return '';
            }
        }

        function appendNum(num) {
            playClickSound(num === '0' ? "C3" : "E4");
            if (currentInput === '0' || currentInput === 'Ошибка') {
                currentInput = num;
            } else {
                if (currentInput.length < 12) {
                    currentInput += num;
                }
            }
            updateDisplay();
        }

        function appendDot() {
            playClickSound("G4");
            if (!currentInput.includes('.')) {
                currentInput += '.';
            }
            updateDisplay();
        }

        function appendOp(op) {
            playClickSound("A4");
            if (currentInput === 'Ошибка') return;
            if (previousInput !== '' && operation) {
                calculate(false);
            }
            operation = op;
            previousInput = currentInput;
            currentInput = '0';
            updateDisplay();
        }

        function clearAll() {
            playClickSound("C2");
            currentInput = '0';
            previousInput = '';
            operation = null;
            updateDisplay();
        }

        function deleteLast() {
            playClickSound("D3");
            if (currentInput.length > 1) {
                currentInput = currentInput.slice(0, -1);
            } else {
                currentInput = '0';
            }
            updateDisplay();
        }

        function calcSqrt() {
            playClickSound("B4");
            let val = parseFloat(currentInput);
            if (val < 0) {
                currentInput = 'Ошибка';
            } else {
                let res = Math.sqrt(val);
                currentInput = Number.isInteger(res) ? res.toString() : res.toFixed(6).replace(/\.?0+$/, '');
            }
            updateDisplay();
        }

        function calculate(playSound = true) {
            if (playSound) playClickSound("E5");
            if (!operation || previousInput === '') return;

            let prev = parseFloat(previousInput);
            let curr = parseFloat(currentInput);
            let result = 0;

            switch(operation) {
                case '+': result = prev + curr; break;
                case '-': result = prev - curr; break;
                case '*': result = prev * curr; break;
                case '/': 
                    if (curr === 0) {
                        currentInput = 'Ошибка';
                        previousInput = '';
                        operation = null;
                        updateDisplay();
                        return;
                    }
                    result = prev / curr; 
                    break;
            }

            // Round precision
            if (!Number.isInteger(result)) {
                result = parseFloat(result.toFixed(8));
            }

            currentInput = result.toString();
            previousInput = '';
            operation = null;
            updateDisplay();
        }

        // Toggle Punk Mode (Visuals + Riff Sound)
        punkToggle.addEventListener('click', () => {
            isPunkMode = !isPunkMode;
            initAudio();
            Tone.start();

            if (isPunkMode) {
                document.querySelector('main').classList.add('punk-active');
                modeStatus.innerText = "ПАНК-РОК РЕЖИМ 🎸";
                modeStatus.className = "text-xs font-bold uppercase tracking-wider text-pink-500 mono-font animate-bounce";
                
                // Play heavy punk chord riff
                synth.triggerAttackRelease(["E2", "B2", "E3"], "4n");
                setTimeout(() => synth.triggerAttackRelease(["G2", "D3", "G3"], "8n"), 250);
                setTimeout(() => synth.triggerAttackRelease(["A2", "E3", "A3"], "4n"), 450);
            } else {
                document.querySelector('main').classList.remove('punk-active');
                modeStatus.innerText = "Оренбург Чилл";
                modeStatus.className = "text-xs font-bold uppercase tracking-wider text-emerald-400 mono-font";
            }
        });

        // Boss Quote Modal
        const bossButton = document.getElementById('bossButton');
        const bossModal = document.getElementById('bossModal');
        const bossQuote = document.getElementById('bossQuote');

        bossButton.addEventListener('click', () => {
            const randomQuote = bossQuotes[Math.floor(Math.random() * bossQuotes.length)];
            bossQuote.innerText = `"${randomQuote}"`;
            bossModal.classList.remove('hidden');
            if (isPunkMode) playClickSound("G5");
        });

        function closeModal() {
            bossModal.classList.add('hidden');
        }

        // Keyboard navigation
        document.addEventListener('keydown', (e) => {
            if (e.key >= '0' && e.key <= '9') appendNum(e.key);
            if (e.key === '.') appendDot();
            if (e.key === '+') appendOp('+');
            if (e.key === '-') appendOp('-');
            if (e.key === '*') appendOp('*');
            if (e.key === '/') appendOp('/');
            if (e.key === 'Enter' || e.key === '=') calculate();
            if (e.key === 'Backspace') deleteLast();
            if (e.key === 'Escape') clearAll();
        });

        // Background subtle canvas particle matrix
        const canvas = document.getElementById('bgCanvas');
        const ctx = canvas.getContext('2d');
        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        resizeCanvas();
        window.addEventListener('resize', resizeCanvas);

        const dots = Array.from({length: 30}, () => ({
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            r: Math.random() * 2 + 1,
            dx: (Math.random() - 0.5) * 0.5,
            dy: (Math.random() - 0.5) * 0.5
        }));

        function animateBg() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = isPunkMode ? '#ff007f' : '#00ff66';

            dots.forEach(d => {
                ctx.beginPath();
                ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
                ctx.fill();

                d.x += d.dx;
                d.y += d.dy;

                if (d.x < 0 || d.x > canvas.width) d.dx *= -1;
                if (d.y < 0 || d.y > canvas.height) d.dy *= -1;
            });

            requestAnimationFrame(animateBg);
        }
        animateBg();
    </script>
